cmake_minimum_required(VERSION 3.21)
# CET macOS. arm64 build of Cyber Engine Tweaks for the native macOS game (2.3.1), loaded as a RED4ext plugin.
# Windows builds keep using xmake.lua; nothing here is read by it. Pattern: let_there_be_flight/macos/CMakeLists.txt.
project(cyber_engine_tweaks LANGUAGES C CXX OBJCXX ASM)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_ARCHITECTURES "arm64")
if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
endif()

if(NOT APPLE)
    message(FATAL_ERROR "This target builds the macOS Cyber Engine Tweaks dylib")
endif()

set(CET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(CET_SRC "${CET_DIR}/src")
set(MAC_REPOS "${CET_DIR}/../.." CACHE PATH "the folder that holds the sibling checkouts")
set(RED4EXT_SDK_DIR "${MAC_REPOS}/RED4ext.SDK-macos" CACHE PATH "macOS RED4ext SDK checkout")
set(CODEWARE_DIR "${MAC_REPOS}/cp2077-codeware" CACHE PATH "macOS Codeware checkout (its SDK rows are reused)")
set(CET_VENDOR_MAC "${CET_DIR}/../vendor-mac" CACHE PATH "dependency sources: luajit2, imgui, sol2, json, hopscotch-map, stb, TiltedCore")
set(SDK_INCLUDE "${RED4EXT_SDK_DIR}/include")
set(HOMEBREW_PREFIX "/opt/homebrew" CACHE PATH "Homebrew prefix (spdlog, fmt, freetype, libpng as static libraries)")

# ---------------------------------------------------------------------------------------------------------------------------------
# Version header (xmake generates src/config/CETVersion.h from src/xmake/CETVersion.h.in; this writes the same defines).
execute_process(COMMAND git -C "${CET_DIR}" describe --tags OUTPUT_VARIABLE CET_GIT_TAG OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git -C "${CET_DIR}" describe --tags --long OUTPUT_VARIABLE CET_GIT_TAG_LONG OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git -C "${CET_DIR}" rev-parse --abbrev-ref HEAD OUTPUT_VARIABLE CET_GIT_BRANCH OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git -C "${CET_DIR}" rev-parse --short HEAD OUTPUT_VARIABLE CET_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git -C "${CET_DIR}" rev-parse HEAD OUTPUT_VARIABLE CET_GIT_COMMIT_LONG OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git -C "${CET_DIR}" log -1 --date=format:%Y%m%d%H%M%S --format=%ad OUTPUT_VARIABLE CET_GIT_COMMIT_DATE
                OUTPUT_STRIP_TRAILING_WHITESPACE)
string(REGEX MATCH "^v?([0-9]+)\\.([0-9]+)\\.([0-9]+)" _ver "${CET_GIT_TAG}")
if(NOT _ver)
    message(FATAL_ERROR "git describe --tags gave '${CET_GIT_TAG}', not vX.Y.Z")
endif()
set(CET_VERSION_MAJOR ${CMAKE_MATCH_1})
set(CET_VERSION_MINOR ${CMAKE_MATCH_2})
set(CET_VERSION_PATCH ${CMAKE_MATCH_3})
string(TIMESTAMP CET_CURRENT_YEAR "%Y")
set(CET_CONFIG_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CET_CONFIG_DIR}/config/CETVersion.h" "#pragma once
// Generated by macos/CMakeLists.txt (the macOS build's copy of src/xmake/CETVersion.h.in).
#define CET_DEBUG_BUILD 0
#define CET_GIT_TAG \"${CET_GIT_TAG}\"
#define CET_GIT_TAG_LONG \"${CET_GIT_TAG_LONG}\"
#define CET_GIT_BRANCH \"${CET_GIT_BRANCH}\"
#define CET_GIT_COMMIT \"${CET_GIT_COMMIT}\"
#define CET_GIT_COMMIT_LONG \"${CET_GIT_COMMIT_LONG}\"
#define CET_GIT_COMMIT_DATE ${CET_GIT_COMMIT_DATE}

#define CET_VERSION_MAJOR ${CET_VERSION_MAJOR}
#define CET_VERSION_MINOR ${CET_VERSION_MINOR}
#define CET_VERSION_PATCH ${CET_VERSION_PATCH}
#define CET_VERSION_SIMPLE \"${CET_VERSION_MAJOR}.${CET_VERSION_MINOR}.${CET_VERSION_PATCH}\"
#define CET_VERSION_FULL \"${CET_GIT_TAG} [${CET_GIT_BRANCH}]\"
#define CET_VERSION CET_VERSION_FULL

#define CET_CURRENT_YEAR ${CET_CURRENT_YEAR}
#define CET_PRODUCT_NAME \"Cyber Engine Tweaks\"
#define CET_GROUP_NAME \"Cyber Engine Tweaks Team\"
#define CET_FILE_NAME \"cyber_engine_tweaks.dylib\"
")

# ---------------------------------------------------------------------------------------------------------------------------------
# LuaJIT (openresty luajit2, GC64 is the arm64 default; Lua 5.2 compatibility as the Windows build).
set(LUAJIT_OUT "${CMAKE_CURRENT_BINARY_DIR}/luajit")
execute_process(
    COMMAND /bin/bash "${CMAKE_CURRENT_SOURCE_DIR}/build-luajit.sh" "${CET_VENDOR_MAC}/luajit2" "${LUAJIT_OUT}"
    RESULT_VARIABLE LUAJIT_RESULT)
if(NOT LUAJIT_RESULT EQUAL 0)
    message(FATAL_ERROR "macos/build-luajit.sh failed (${LUAJIT_RESULT})")
endif()

# TiltedCore's headers are included as <TiltedCore/...>.
set(TILTED_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/tiltedcore-include")
if(NOT EXISTS "${TILTED_INCLUDE}/TiltedCore")
    file(MAKE_DIRECTORY "${TILTED_INCLUDE}")
    file(CREATE_LINK "${CET_VENDOR_MAC}/TiltedCore/Code/core/include" "${TILTED_INCLUDE}/TiltedCore" SYMBOLIC)
endif()

# ---------------------------------------------------------------------------------------------------------------------------------
# Generated SDK overlay (build/sdk-overlay, first on the include path): Relocation-inl.hpp = the Mac SDK table plus Codeware's
# MacSdkRows.inc (SDK-core rows) plus macos/CetSdkRows.inc (CET's rows). An unknown hash goes to cet_mac_trap_for_hash
# (macos/MacShims.cpp): a stub that logs once and returns 0. Nothing ever resolves to 0, and no hook is ever placed on the stub.
set(SDK_OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/sdk-overlay")
set(SDK_RELOC "${SDK_INCLUDE}/RED4ext/Relocation-inl.hpp")
set(CW_ROWS_INC "${CODEWARE_DIR}/macos/MacSdkRows.inc")
set(CET_ROWS_INC "${CMAKE_CURRENT_SOURCE_DIR}/CetSdkRows.inc")
file(READ "${SDK_RELOC}" SDK_RELOC_TEXT)
file(READ "${CW_ROWS_INC}" CW_ROWS_TEXT)
file(READ "${CET_ROWS_INC}" CET_ROWS_TEXT)
set(SDK_ROWS_MARKER "        { 4201325776, 0x109BAA8 },  // CGlobalFunction_ctor\n")
string(FIND "${SDK_RELOC_TEXT}" "${SDK_ROWS_MARKER}" SDK_ROWS_POS)
if(SDK_ROWS_POS EQUAL -1)
    message(FATAL_ERROR "the anchor row (CGlobalFunction_ctor) is not in ${SDK_RELOC}")
endif()
string(REPLACE "${SDK_ROWS_MARKER}" "${SDK_ROWS_MARKER}        // --- cp2077-codeware/macos/MacSdkRows.inc ---\n${CW_ROWS_TEXT}        // --- CyberEngineTweaks/macos/CetSdkRows.inc ---\n${CET_ROWS_TEXT}" SDK_RELOC_TEXT "${SDK_RELOC_TEXT}")
set(SDK_TAIL_START "    // Address not found or placeholder - return a non-zero sentinel value")
set(SDK_TAIL_END "    return 0;\n#else\n    if constexpr (Detail::AddressResolverOverride<uint32_t>::value)")
string(FIND "${SDK_RELOC_TEXT}" "${SDK_TAIL_START}" SDK_TAIL_A)
string(FIND "${SDK_RELOC_TEXT}" "${SDK_TAIL_END}" SDK_TAIL_B)
if(SDK_TAIL_A EQUAL -1 OR SDK_TAIL_B EQUAL -1)
    message(FATAL_ERROR "the unknown-hash tail of Resolve was not found in ${SDK_RELOC}")
endif()
string(SUBSTRING "${SDK_RELOC_TEXT}" 0 ${SDK_TAIL_A} SDK_HEAD)
math(EXPR SDK_REST_FROM "${SDK_TAIL_B} + 14")
string(SUBSTRING "${SDK_RELOC_TEXT}" ${SDK_REST_FROM} -1 SDK_REST)
set(SDK_RELOC_TEXT "${SDK_HEAD}    // CET macOS: fail-safe trap. Logs the hash once, returns a stub that logs its first call and returns 0.\n    return cet_mac_trap_for_hash(aHash);\n${SDK_REST}")
set(SDK_DECL_MARKER "extern \"C\" uintptr_t tweakxl_class_vslot(uint32_t aByteOffset);\n")
string(FIND "${SDK_RELOC_TEXT}" "${SDK_DECL_MARKER}" SDK_DECL_POS)
if(SDK_DECL_POS EQUAL -1)
    message(FATAL_ERROR "the extern declaration anchor was not found in ${SDK_RELOC}")
endif()
string(REPLACE "${SDK_DECL_MARKER}" "${SDK_DECL_MARKER}extern \"C\" uintptr_t cet_mac_trap_for_hash(uint32_t aHash);\n" SDK_RELOC_TEXT "${SDK_RELOC_TEXT}")
file(WRITE "${SDK_OVERLAY}/RED4ext/Relocation-inl.hpp" "${SDK_RELOC_TEXT}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SDK_RELOC}" "${CW_ROWS_INC}" "${CET_ROWS_INC}")

# ---------------------------------------------------------------------------------------------------------------------------------
# Sources: all of src/ except the Windows renderer, the Win32/DX12 ImGui backends, the WndProc window and the byte patches; their
# macOS counterparts are src/metal, src/imgui_impl/osx.mm and src/window/window_mac.mm.
file(GLOB_RECURSE CET_SOURCES CONFIGURE_DEPENDS "${CET_SRC}/*.cpp" "${CET_SRC}/*.mm")
list(FILTER CET_SOURCES EXCLUDE REGEX "/src/d3d12/")
list(FILTER CET_SOURCES EXCLUDE REGEX "/src/patches/")
list(REMOVE_ITEM CET_SOURCES "${CET_SRC}/reverse/RenderContext.cpp" "${CET_SRC}/imgui_impl/dx12.cpp" "${CET_SRC}/imgui_impl/win32.cpp" "${CET_SRC}/window/window.cpp")

set(IMGUI_DIR "${CET_VENDOR_MAC}/imgui")
set(IMGUI_SOURCES
    "${IMGUI_DIR}/imgui.cpp"
    "${IMGUI_DIR}/imgui_demo.cpp"
    "${IMGUI_DIR}/imgui_draw.cpp"
    "${IMGUI_DIR}/imgui_tables.cpp"
    "${IMGUI_DIR}/imgui_widgets.cpp"
    "${IMGUI_DIR}/misc/freetype/imgui_freetype.cpp"
    "${IMGUI_DIR}/backends/imgui_impl_metal.mm"
)
set(TILTED_DIR "${CET_VENDOR_MAC}/TiltedCore/Code/core/src")
set(TILTED_SOURCES
    "${TILTED_DIR}/Allocator.cpp"
    "${TILTED_DIR}/BoundedAllocator.cpp"
    "${TILTED_DIR}/Buffer.cpp"
    "${TILTED_DIR}/Hash.cpp"
    "${TILTED_DIR}/Math.cpp"
    "${TILTED_DIR}/ScratchAllocator.cpp"
    "${TILTED_DIR}/Serialization.cpp"
    "${TILTED_DIR}/StandardAllocator.cpp"
    "${TILTED_DIR}/TaskQueue.cpp"
    "${TILTED_DIR}/ViewBuffer.cpp"
)
set(CET_MAC_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/HvfStubs.c"
    MacShims.cpp
    MacEntry.cpp
    MacVault.cpp
    TiltedMallocAllocator.cpp
    OverrideThunks.S
)

add_library(cyber_engine_tweaks SHARED ${CET_SOURCES} ${CET_MAC_SOURCES} ${IMGUI_SOURCES} ${TILTED_SOURCES})
set_target_properties(cyber_engine_tweaks PROPERTIES OUTPUT_NAME "cyber_engine_tweaks" PREFIX "" SUFFIX ".dylib")

set(IMGUI_USER_CONFIG "${CET_SRC}/imgui_impl/imgui_user_config.h")
target_compile_definitions(cyber_engine_tweaks PRIVATE
    RED4EXT_PLATFORM_MACOS RED4EXT_HEADER_ONLY
    NOMINMAX SOL_NO_NIL=0 SOL_ALL_SAFETIES_ON SOL_LUAJIT=1 SOL_EXCEPTIONS_SAFE_PROPAGATION
    IMGUI_USER_CONFIG="${IMGUI_USER_CONFIG}" IMGUI_ENABLE_FREETYPE IMGUI_USE_WCHAR32
    SPDLOG_FMT_EXTERNAL
)

# Codeware's macOS SDK header fixes (gameIGameSystem: IsSavingLocked returns its SaveLock through x8 on arm64), ahead of the SDK.
# An SDK on the current upstream API (Containers/DynArray.hpp) already has them, and the copies would shadow its headers.
if(EXISTS "${SDK_INCLUDE}/RED4ext/Containers/DynArray.hpp")
    set(CET_CODEWARE_SDK_FIXES "")
else()
    set(CET_CODEWARE_SDK_FIXES "${CODEWARE_DIR}/macos/inc")
endif()

target_include_directories(cyber_engine_tweaks PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/inc"
    "${CMAKE_CURRENT_SOURCE_DIR}"
    ${CET_CODEWARE_SDK_FIXES}
    "${SDK_OVERLAY}"
    "${SDK_INCLUDE}"
    "${CET_SRC}"
    "${CET_CONFIG_DIR}"
    "${LUAJIT_OUT}/include"
    "${IMGUI_DIR}"
    "${IMGUI_DIR}/backends"
    "${IMGUI_DIR}/misc/freetype"
    "${CET_VENDOR_MAC}/sol2/include"
    "${CET_VENDOR_MAC}/json/single_include"
    "${CET_VENDOR_MAC}/hopscotch-map/include"
    "${CET_VENDOR_MAC}/stb"
    "${TILTED_INCLUDE}"
    "${CET_VENDOR_MAC}/TiltedCore/Code/core/include"
    "${HOMEBREW_PREFIX}/include"
    "${HOMEBREW_PREFIX}/include/freetype2"
)

# Game symbols are reached through address hashes, never by name; check-undefined.sh proves nothing else is left undefined.
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -undefined dynamic_lookup")
target_link_libraries(cyber_engine_tweaks PRIVATE
    "${LUAJIT_OUT}/lib/libluajit-5.1.a"
    "${HOMEBREW_PREFIX}/lib/libspdlog.a"
    "${HOMEBREW_PREFIX}/lib/libfmt.a"
    "${HOMEBREW_PREFIX}/lib/libfreetype.a"
    "${HOMEBREW_PREFIX}/lib/libpng16.a"
    "${HOMEBREW_PREFIX}/lib/libbrotlidec.a"
    "${HOMEBREW_PREFIX}/lib/libbrotlicommon.a"
    sqlite3 z bz2
    "-framework Foundation" "-framework AppKit" "-framework QuartzCore" "-framework Metal" "-framework CoreGraphics"
)

target_precompile_headers(cyber_engine_tweaks PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${CET_SRC}/stdafx.h>)
set_source_files_properties(${IMGUI_SOURCES} ${TILTED_SOURCES} MacVault.cpp TiltedMallocAllocator.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
# ImGui's Metal backend defines Objective-C classes with generic names; NightCity Console's overlay (injected in the same process) carries
# its own copy, and two classes with one name make the Objective-C runtime pick one at random. CET's copies get their own names.
set_source_files_properties("${IMGUI_DIR}/backends/imgui_impl_metal.mm" PROPERTIES COMPILE_DEFINITIONS
    "MetalBuffer=CETImGuiMetalBuffer;FramebufferDescriptor=CETImGuiFramebufferDescriptor;MetalContext=CETImGuiMetalContext")

target_compile_options(cyber_engine_tweaks PRIVATE
    -fexceptions
    -fms-extensions
    -Wno-deprecated-declarations
    -Wno-microsoft-template
    -Wno-invalid-offsetof
    -Wno-nullability-completeness
    -fvisibility=hidden
    -fvisibility-inlines-hidden
    $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>
)

get_filename_component(_repo_root "${CET_DIR}" ABSOLUTE)
get_filename_component(_sdk_root "${RED4EXT_SDK_DIR}" ABSOLUTE)
get_filename_component(_vendor_root "${CET_VENDOR_MAC}" ABSOLUTE)
target_compile_options(cyber_engine_tweaks PRIVATE "-ffile-prefix-map=${_repo_root}=." "-ffile-prefix-map=${_sdk_root}=RED4ext.SDK-macos"
                       "-ffile-prefix-map=${_vendor_root}=vendor-mac" "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")

add_custom_command(TARGET cyber_engine_tweaks POST_BUILD
    COMMAND codesign -f -s - "$<TARGET_FILE:cyber_engine_tweaks>"
    COMMAND /bin/bash "${CMAKE_CURRENT_SOURCE_DIR}/check-undefined.sh" "$<TARGET_FILE:cyber_engine_tweaks>"
    VERBATIM
)

message(STATUS "Cyber Engine Tweaks ${CET_GIT_TAG} (${CET_GIT_BRANCH} ${CET_GIT_COMMIT}) configured for macOS")
